  (function () {
    var body = document.body;

      "(min-width: 768px) and (max-width: 1439px)",

    var guideStorageKey = "agmNavGuideDismissedAt";
    var pinStorageKey = "agmNavPinned";
    var guideCacheMs = 24 * 60 * 60 * 1000;

    if (!drawer || !overlay || !openBtn || !closeBtn) {
      return;
    }

    function isPinnedStored() {
      try {
        return localStorage.getItem(pinStorageKey) === "1";
      } catch (e) {
        return false;
      }
    }

    function applyPinnedState(isPinned) {
      body.classList.toggle("agm-nav-pinned", isPinned);
      if (pinBtn) {
        pinBtn.setAttribute("aria-pressed", isPinned ? "true" : "false");
        pinBtn.setAttribute(
          "title",
          isPinned ? "取消钉住侧栏" : "钉住侧栏展开",
        );
        pinBtn.setAttribute(
          "aria-label",
          isPinned ? "取消钉住侧栏" : "钉住侧栏展开",
        );
      }
    }

    function persistPinnedState(isPinned) {
      try {
        localStorage.setItem(pinStorageKey, isPinned ? "1" : "0");
      } catch (e) {}
    }

    function syncCompactPinState() {
      if (compactMq.matches) {
        applyPinnedState(isPinnedStored());
      } else {
        applyPinnedState(false);
      }
    }

    function togglePinnedState() {
      if (!compactMq.matches) {
        return;
      }
      var nextPinned = !body.classList.contains("agm-nav-pinned");
      applyPinnedState(nextPinned);
      persistPinnedState(nextPinned);
    }

    function isGuideWithinCache() {
      try {
        var raw = localStorage.getItem(guideStorageKey);
        if (!raw) return false;
        var dismissedAt = parseInt(raw, 10);
        if (!dismissedAt || isNaN(dismissedAt)) {
          localStorage.removeItem(guideStorageKey);
          return false;
        }
        if (Date.now() - dismissedAt < guideCacheMs) {
          return true;
        }
        localStorage.removeItem(guideStorageKey);
        return false;
      } catch (e) {
        return false;
      }
    }

    function dismissGuide() {
      if (!guide) return;
      try {
        localStorage.setItem(guideStorageKey, String(Date.now()));
      } catch (e) {}
      guide.hidden = true;
    }

    function updateGuideVisibility() {
      if (!guide) return;
      // 手机端不展示导航引导（遮挡内容，且不支持 Ctrl+K）
      guide.hidden = true;
    }

    function setNavOpen(isOpen) {
      body.classList.toggle("agm-nav-open", isOpen);
      openBtn.setAttribute("aria-expanded", isOpen ? "true" : "false");
      if (isOpen) {
        dismissGuide();
      }
    }

    function openNav() {
      if (!drawerMq.matches) return;
      setNavOpen(true);
    }

    function closeNav() {
      setNavOpen(false);

    openBtn.addEventListener("click", openNav);
    closeBtn.addEventListener("click", closeNav);
    overlay.addEventListener("click", closeNav);

    if (guideCloseBtn) {
      guideCloseBtn.addEventListener("click", function (e) {
        e.stopPropagation();
        dismissGuide();
      });
    }

    document.addEventListener("keydown", function (e) {
      if (e.key === "Escape") {
        if (
          window.KeyboardSearch &&
          typeof window.KeyboardSearch.isOpen === "function" &&
          window.KeyboardSearch.isOpen()
        ) {
          return;
        }
        closeNav();
      }

      if ((e.ctrlKey || e.metaKey) && (e.key === "k" || e.key === "K")) {
        dismissGuide();
      }
    });

    if (pinBtn) {
      pinBtn.addEventListener("click", function (e) {
        e.stopPropagation();
        togglePinnedState();
      });
    }

    drawerMq.addEventListener("change", function (e) {
      if (!e.matches) {
        closeNav();
      }
      updateGuideVisibility();
      syncCompactPinState();
    });

    compactMq.addEventListener("change", syncCompactPinState);

    updateGuideVisibility();
    syncCompactPinState();
  })();
